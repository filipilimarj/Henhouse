function getOrCreateSheet_(name, headers) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName(name); if (!sheet) { sheet = ss.insertSheet(name); sheet.appendRow(headers); return sheet; } var existingHeaders = sheet.getRange(1, 1, 1, Math.max(sheet.getLastColumn(), 1)).getValues()[0]; headers.forEach(function (h, i) { if (existingHeaders[i] !== h) { sheet.getRange(1, i + 1).setValue(h); } }); return sheet; } function sheetToObjects_(sheet) { var data = sheet.getDataRange().getValues(); if (data.length < 2) return []; var headers = data[0]; return data.slice(1) .filter(function (row) { return row[0] !== '' && row[0] !== null; }) .map(function (row) { var obj = {}; headers.forEach(function (h, i) { obj[h] = row[i]; }); return obj; }); } function findRowIndexById_(sheet, id) { var data = sheet.getDataRange().getValues(); for (var i = 1; i < data.length; i++) { if (String(data[i][0]) === String(id)) return i + 1; } return -1; } function deleteRowsByHeaderValue_(sheet, headerName, value) { var lastCol = sheet.getLastColumn(); if (lastCol === 0) return 0; var headers = sheet.getRange(1, 1, 1, lastCol).getValues()[0]; var columnIndex = headers.indexOf(headerName); if (columnIndex === -1) return 0; var data = sheet.getDataRange().getValues(); var deletedCount = 0; for (var i = data.length - 1; i >= 1; i--) { if (String(data[i][columnIndex]) === String(value)) { sheet.deleteRow(i + 1); deletedCount++; } } return deletedCount; } function markSalesAsDeposited_(sheet, employeeId, depositId) { var lastCol = sheet.getLastColumn(); if (lastCol === 0) return 0; var headers = sheet.getRange(1, 1, 1, lastCol).getValues()[0]; var empCol = headers.indexOf('funcionarioId'); var depCol = headers.indexOf('depositoId'); if (empCol === -1 || depCol === -1) return 0; var data = sheet.getDataRange().getValues(); var markedCount = 0; for (var i = 1; i < data.length; i++) { var isSameEmployee = String(data[i][empCol]) === String(employeeId); var isPending = !data[i][depCol]; if (isSameEmployee && isPending) { sheet.getRange(i + 1, depCol + 1).setValue(depositId); markedCount++; } } return markedCount; } function getOrInitStockSheet_() { var stockSheet = getOrCreateSheet_('EstoqueGeladeira', ['id', 'nome', 'quantidade', 'lote']); var data = stockSheet.getDataRange().getValues(); if (data.length <= 1) { Object.keys(YIELDS_PADRAO).forEach(function(prato, idx) { stockSheet.appendRow([idx + 1, prato, 0, YIELDS_PADRAO[prato]]); }); } return stockSheet; } function getOrInitInsumosSheet_() { var insumosSheet = getOrCreateSheet_('EstoqueInsumos', ['id', 'nome', 'quantidade']); var data = insumosSheet.getDataRange().getValues(); if (data.length <= 1) { CATALOGO_PADRAO.forEach(function(nome, idx) { insumosSheet.appendRow([idx + 1, nome, 0]); }); } return insumosSheet; } function processSaleStock_(items) { var stockSheet = getOrInitStockSheet_(); var rows = stockSheet.getDataRange().getValues(); if (rows.length <= 1) return; var headers = rows[0]; var nameCol = headers.indexOf('nome'); var qtdCol = headers.indexOf('quantidade'); var loteCol = headers.indexOf('lote'); items.forEach(function(item) { var itemName = item.name || item.nome; var qtdSold = Number(item.quantity || item.qty || 1); for (var i = 1; i < rows.length; i++) { if (String(rows[i][nameCol]).trim().toLowerCase() === String(itemName).trim().toLowerCase()) { var currentQtd = Number(rows[i][qtdCol]) || 0; var batchSize = Number(rows[i][loteCol]) || 1; if (currentQtd >= qtdSold) { currentQtd -= qtdSold; } else { var needed = qtdSold - currentQtd; var batchesToCook = Math.ceil(needed / batchSize); var totalProduced = batchesToCook * batchSize; currentQtd = (currentQtd + totalProduced) - qtdSold; } stockSheet.getRange(i + 1, qtdCol + 1).setValue(currentQtd); rows[i][qtdCol] = currentQtd; break; } } }); } function jsonOutput_(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } function doGet(e) { var produtosSheet = getOrCreateSheet_('Produtos', ['id', 'nome', 'preco', 'imagem']); var funcSheet = getOrCreateSheet_('Funcionarios', ['id', 'nome', 'comissao']); var vendasSheet = getOrCreateSheet_('Vendas', ['id', 'datahora', 'funcionarioId', 'funcionarioNome', 'itens', 'total', 'itensCount', 'data', 'depositoId']); var depositosSheet = getOrCreateSheet_('Depositos', ['id', 'dataHora', 'funcionarioId', 'funcionarioNome', 'totalVendas', 'comissaoPercentual', 'valorFuncionario', 'valorEmpresa', 'qtdVendas']); var stockSheet = getOrInitStockSheet_(); var insumosSheet = getOrInitInsumosSheet_(); var produtos = sheetToObjects_(produtosSheet).map(function (p) { return { id: String(p.id), name: p.nome, price: Number(p.preco), image: p.imagem }; }); var funcionarios = sheetToObjects_(funcSheet).map(function (f) { var rate = Number(f.comissao); return { id: String(f.id), name: f.nome, commissionRate: isNaN(rate) ? 5 : rate }; }); var vendas = sheetToObjects_(vendasSheet).map(function (v) { var itens = []; try { itens = JSON.parse(v.itens); } catch (err) { itens = []; } return { id: v.id, time: v.datahora, date: v.data || '', employeeId: String(v.funcionarioId), employeeName: v.funcionarioNome, items: itens, total: Number(v.total), itemsCount: Number(v.itensCount), depositId: v.depositoId ? String(v.depositoId) : null }; }); var deposits = sheetToObjects_(depositosSheet).map(function (d) { return { id: String(d.id), dateTime: d.dataHora, employeeId: String(d.funcionarioId), employeeName: d.funcionarioNome, totalSales: Number(d.totalVendas), commissionRate: Number(d.comissaoPercentual), employeeCut: Number(d.valorFuncionario), companyCut: Number(d.valorEmpresa), salesCount: Number(d.qtdVendas) }; }); var stock = sheetToObjects_(stockSheet).map(function (s) { return { id: String(s.id), name: s.nome, quantity: Number(s.quantidade) || 0, batchSize: Number(s.lote) || 1 }; }); var insumos = sheetToObjects_(insumosSheet).map(function (i) { return { id: String(i.id), name: i.nome, quantity: Number(i.quantidade) || 0 }; }); return jsonOutput_({ products: produtos, employees: funcionarios, sales: vendas, deposits: deposits, stock: stock, insumos: insumos }); } function doPost(e) { var body; try { body = JSON.parse(e.postData.contents); } catch (err) { return jsonOutput_({ ok: false, error: 'JSON inválido' }); } var lock = LockService.getScriptLock(); var acquired = lock.tryLock(30000); if (!acquired) { return jsonOutput_({ ok: false, error: 'Sistema ocupado. else insumosSheet.getRange(idxIns, 1, 1, rowIns.length).setValues([rowIns]); } else if (action === 'deleteInsumo') { var idxDel = findRowIndexById_(insumosSheet, body.id); if (idxDel !== -1) insumosSheet.deleteRow(idxDel); } else if (action === 'confirmDeposit') { var dep = body.deposit; var rowDep = [dep.id, dep.dateTime, dep.employeeId, dep.employeeName, dep.totalSales, dep.commissionRate, dep.employeeCut, dep.companyCut, dep.salesCount]; depositosSheet.appendRow(rowDep); var markedCount = markSalesAsDeposited_(vendasSheet, dep.employeeId, dep.id); return jsonOutput_({ ok: true, salesMarked: markedCount }); } else { return jsonOutput_({ ok: false, error: 'Ação desconhecida: ' + action }); } return jsonOutput_({ ok: true }); } catch (err) { return jsonOutput_({ ok: false, error: String(err) }); } finally { lock.releaseLock(); } }
